docker images			          check docker images
docker ps			              check running docker containers
docker ps -a			          check previousely run docker containers
docker stats			          stats for running containers

docker save dumidu1 > dumidu1.tar	                        save a docker image
docker load < dumidu1.tar		                              load a docker image

docker run hello-world			                              run a docker image

docker stop [container-name]	                            stop a running container
docker start [image-name]	                                start a stopped container

docker rm [container_name]	                              remove running container
docker rmi dumidu1		                                    remove docker image
docker rmi --force dumidu1	                              forcefully remove docker image
